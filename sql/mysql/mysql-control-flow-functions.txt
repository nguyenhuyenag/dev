
- Các hàm họ IF trong MySQL
		+------------------------+-----------------------------------------
		| IF(đk, a, b)           | Rẽ nhánh 2 giá trị                      
		| IFNULL(a, b)           | Nếu a là NULL → trả b, ngược lại -> trả a
		| NULLIF(a, b)           | Nếu a = b → trả NULL, ngược lại trả a  
		| COALESCE(a, b, c, ...) | Trả giá trị không NULL đầu tiên        
		+------------------------+-----------------------------------------
	
	- Dùng để xử lý NULL
	
		SELECT 
		  u.name,
		  IF(SUM(r.distance) IS NULL, 0, SUM(r.distance)) AS travelled_distance
		FROM Users u
		LEFT JOIN Rides r 
			ON u.id = r.user_id
		GROUP BY u.id, u.name;
	
		-- Ví dụ: hiển thị số điện thoại, nếu NULL thì ghi 'Chưa có'
		SELECT 
			name,
			IFNULL(phone, 'Chưa có') 
		AS phone FROM users;

		-- Ví dụ: tránh chia cho 0
		SELECT total / NULLIF(count, 0) AS avg FROM stats;
	
- CASE

	1.Simple CASE
			SELECT
				col1,
				...
				CASE col2
					WHEN <giá_trị_1> THEN <kết_quả_1>
					WHEN <giá_trị_2> THEN <kết_quả_2>
					[ELSE <kết_quả_mặc_định>]
				END
			FROM <bảng>;
		Ví dụ:
			SELECT
			  CASE t.gender
				WHEN 1 THEN 'Nam'
				WHEN 2 THEN 'Nữ'
				ELSE 'Không xác định'
			  END AS 'gender'
			FROM Users t;
	2. Searched CASE
		SELECT
		  CASE
			WHEN t.distance >= 300 THEN 'Cao'
			WHEN t.distance >= 150 THEN 'Trung bình'
			WHEN t.distance > 0    THEN 'Thấp'
			ELSE 'Không đi'
		  END AS muc_do
		FROM Rides t;
